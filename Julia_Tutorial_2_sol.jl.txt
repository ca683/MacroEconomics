# Let's create some data to plot
x = 1:10
y = x.^3
y1 = x.^2

plot(x, y, label="x^3", linewidth = 3, legend = :topleft) # To create a LaTeX string, we have to write L before the string and put the LaTeX code inside $
plot!(x, y1, label="x^2", linewidth = 3)
xlabel!("Number of chocolate bars")
ylabel!("Weight")
title!("Obesity Figure 1")

# A second way of plotting
plot(x, [y, y1],
   label = ["x^3" "x^2"],
   linewidth = 3,
   legend = :topleft, # or no legend, legend = false
   xlabel = "Number of chocolate bars",
   ylabel = "Weight",
   title = "Obesity Figure 2")

# For Loops
# Example: Take for example depreciation, we could just type out the sequence of calculations above by hand,
# one after the other.
δ = 0.143;  # Rate of depreciation (the greek letter δ is a Unicode character. You get it by typing \delta and clicking the TAB button)
m = 4;  # Number of rows
n = 1;  # Number of columns
K = zeros(m,n); # Prepopulate a matrix where we will store our results
K[1] = 0.5;
K[2] = (1-δ)*K[1];
K[3] = (1-δ)*K[2];
K[4] = (1-δ)*K[3];

# The for loop provides a much more efficient way of doing this calculation:
δ = 0.143;  # Rate of depreciation
m = 4;  # Number of rows
n = 1;  # Number of columns
K = zeros(m,n); # Prepopulate a matrix where we will store our results
K[1] = 0.5;
for k = 2:4
   K[k] = (1-δ)*K[k-1]; # Notice it doesn't work if it is not indented
end


# An equivalent way of writing the loop above
δ = 0.143;  # Rate of depreciation
m = 4;  # Number of rows
n = 1;  # Number of columns
K = zeros(m,n); # Prepopulate a matrix where we will store our results
K[1] = 0.5;
for k=1:3
   K[k+1] = (1-δ)*K[k];
end


# The increment in a for loop doesn't have to be 1
for j = 2:2:8
   println(3*j)
end


# For loops are not restricted to a single operation
x = 4:8;
m = 5;  # Number of rows
n = 1;  # Number of columns
s = zeros(m,n); # Prepopulate a matrix where we will store our results
c = zeros(m,n); # Prepopulate a matrix where we will store our results
f = zeros(m,n); # Prepopulate a matrix where we will store our results
for k=1:m
   s[k] = x[k]^2;
   c[k] = x[k]^3;
   f[k] = x[k]^4;
end

# Now a longer For Loop. Notice this would be a pain by hand.
δ = 0.143;
t = 0:25;
m = 26;  # Number of rows
n = 1;  # Number of columns
r = zeros(m,n); # Prepopulate a matrix where we will store our results
r[1] = 1;
for k = 2:m;
   r[k] = (1-δ)*r[k-1];
end
plot(t,r)
xlabel!("Time period")
ylabel!("K(t)/K(0)")


# Notice we can nest loops
δ = 0.143;
t = 0:25;
m = 26;  # Number of rows
n = 5;  # Number of columns
r = zeros(m,n); # Prepopulate a matrix where we will store our results
r[1,:] .= 1;
for k = 2 : m
   # @show k # @show command allows you to see output in each iteration
   for j = 2 : n
      r[k,j] = (1-δ)*r[k-1,j-1]*j
   end
end
plot(t,r)

# Another way to allocate an array and fill it is to use a push! command
c = []
for i = 1:5
   push!(c,2*i)
end

# This can equivalently be done in one line by using an "array comprehension"
t = [2*i for i=1:5]

c == t

# Exercise 6
# Suppose that you have a $20,000 loan at 6 percent annual interest compounded monthly.
# You can afford to pay $200/month toward this loan. How long will it take until the loan
# is paid off? To do this calculation, think of the amount remaining on the loan from time
# k to time k+1 as a combination of interest accrued and payment made, express this as an iteration,
# and then write code using a for loop to find the amount remaining in the loan after N months.
# You may need to experiment with the value of N to find the value in which the remaining amount passes through zero (and the loan is paid off).
# Plot the amount remaining in the loan as a function of N. Label the axes, and label the legend appropriately, and include a title.

interest= 0.06/12;
t =1:140;
m = 140;  # Number of rows
n = 1;  # Number of columns
r = zeros(m,n); # Create matrix U and fill it with zeros
r[1] = 20000;
for k = 2:m;
   r[k] = (1+interest)*r[k-1] -200;
end
plot(t,r , label = "loan", title = "Loan amortisation chart")
xlabel!("Number of months")
ylabel!("Remaining Loan amount")
